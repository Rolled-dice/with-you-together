const asmrAudio = {
  ctx: null,
  init() {
    if (typeof window === 'undefined') return null;
    if (!this.ctx) {
      const AudioCtx = window.AudioContext || window.webkitAudioContext;
      if (AudioCtx) this.ctx = new AudioCtx();
    }
    if (this.ctx && this.ctx.state === 'suspended') this.ctx.resume();
    return this.ctx;
  },
  playSoftChime(pitchMultiplier = 1) {
    try {
      const ctx = this.init();
      if (!ctx) return;
      const now = ctx.currentTime;
      const osc = ctx.createOscillator();
      const gain = ctx.createGain();
      osc.type = 'sine';
      osc.frequency.setValueAtTime(528 * pitchMultiplier, now);
      gain.gain.setValueAtTime(0.06, now);
      gain.gain.exponentialRampToValueAtTime(0.0008, now + 0.45);
      osc.connect(gain);
      gain.connect(ctx.destination);
      osc.start(now);
      osc.stop(now + 0.45);
    } catch {}
  },
  playBubblePop(freq = 420) {
    try {
      const ctx = this.init();
      if (!ctx) return;
      const now = ctx.currentTime;
      const osc = ctx.createOscillator();
      const gain = ctx.createGain();
      osc.type = 'triangle';
      osc.frequency.setValueAtTime(freq, now);
      osc.frequency.exponentialRampToValueAtTime(freq * 1.9, now + 0.08);
      gain.gain.setValueAtTime(0.11, now);
      gain.gain.exponentialRampToValueAtTime(0.001, now + 0.09);
      osc.connect(gain);
      gain.connect(ctx.destination);
      osc.start(now);
      osc.stop(now + 0.09);
    } catch {}
  },
  playSingingBowl() {
    try {
      const ctx = this.init();
      if (!ctx) return;
      const now = ctx.currentTime;
      [432, 648, 864].forEach((freq, idx) => {
        const osc = ctx.createOscillator();
        const gain = ctx.createGain();
        osc.type = 'sine';
        osc.frequency.setValueAtTime(freq, now);
        gain.gain.setValueAtTime(0.07 / (idx + 1), now);
        gain.gain.exponentialRampToValueAtTime(0.0005, now + 2.2);
        osc.connect(gain);
        gain.connect(ctx.destination);
        osc.start(now);
        osc.stop(now + 2.2);
      });
    } catch {}
  }
};

// 7. THREE.JS 3D FAMILY GLOBE STAGE (`ThreeJsFamilyStage`)
// ============================================================================
export const ThreeJsFamilyStage = ({
  members = INITIAL_FAMILY_MEMBERS,
  zoom = 1,
  setZoom = () => {},
  autoRotate = true,
  setAutoRotate = () => {},
  expandedNotifId = null,
  setExpandedNotifId = () => {},
  targetRotationYaw = null,
  onUpdateMemberOptIn = null,
  globeConfig = {
    theme: 'navy',
    rotationSpeed: 0.0032,
    showPulseRings: true
  }
}) => {
  const mountRef = useRef(null);
  const [projectedNodes, setProjectedNodes] = useState([]);
  const rotationRef = useRef({ yaw: 0.35, pitch: 0.12 });
  const dragRef = useRef({ isDragging: false, lastX: 0, lastY: 0 });
  const membersRef = useRef(members);
  const zoomRef = useRef(zoom);
  const autoRotateRef = useRef(autoRotate);
  const globeConfigRef = useRef(globeConfig);

  useEffect(() => {
    membersRef.current = members;
  }, [members]);
  useEffect(() => {
    zoomRef.current = zoom;
  }, [zoom]);
  useEffect(() => {
    autoRotateRef.current = autoRotate;
  }, [autoRotate]);
  useEffect(() => {
    globeConfigRef.current = globeConfig;
  }, [globeConfig]);
  useEffect(() => {
    if (typeof targetRotationYaw === 'number') {
      rotationRef.current.yaw = targetRotationYaw;
    }
  }, [targetRotationYaw]);

  useEffect(() => {
    const container = mountRef.current;
    if (!container) return;

    const canvas = document.createElement('canvas');
    canvas.style.width = '100%';
    canvas.style.height = '100%';
    canvas.style.display = 'block';
    container.innerHTML = '';
    container.appendChild(canvas);

    const ctx = canvas.getContext('2d');
    if (!ctx) return;

    let width = container.clientWidth || 540;
    let height = container.clientHeight || 400;
    const dpr = Math.min(window.devicePixelRatio || 1, 2);

    const scene = new THREE.Scene();
    const camera = new THREE.PerspectiveCamera(42, width / height, 0.1, 100);
    camera.position.set(0, 0, 4.3);
    camera.lookAt(0, 0, 0);
    camera.updateMatrixWorld(true);

    const globeGroup = new THREE.Group();
    scene.add(globeGroup);

    const syncCanvasSize = () => {
      width = Math.max(280, Number(container.clientWidth) || 540);
      height = Math.max(280, Number(container.clientHeight) || 400);
      canvas.width = width * dpr;
      canvas.height = height * dpr;
      camera.aspect = width / height;
      camera.updateProjectionMatrix();
      camera.updateMatrixWorld(true);
    };
    syncCanvasSize();

    const dotCount = 1800;
    const dotItems = [];
    const goldenRatio = (1 + Math.sqrt(5)) / 2;
    for (let i = 0; i < dotCount; i++) {
      const y = 1 - (i / (dotCount - 1)) * 2;
      const radiusAtY = Math.sqrt(1 - y * y);
      const theta = (2 * Math.PI * i) / goldenRatio;
      const x = Math.cos(theta) * radiusAtY;
      const z = Math.sin(theta) * radiusAtY;
      const mask = Math.sin(x * 4.2 + 0.5) * Math.cos(y * 3.8) + Math.sin(z * 4.5 - 0.3) * 0.6;
      const isLand = mask > -0.12;
      const r = isLand ? 1.245 : 1.22;
      dotItems.push({
        localPos: new THREE.Vector3(x * r, y * r, z * r),
        mask,
        size: isLand ? 2.3 : 1.3
      });
    }

    const sphericalToVec3 = (phi, theta, radius = 1.28) =>
      new THREE.Vector3(
        radius * Math.sin(phi) * Math.sin(theta),
        radius * Math.cos(phi),
        radius * Math.sin(phi) * Math.cos(theta)
      );

    const project3DToScreen = (localVec) => {
      const worldPos = localVec.clone().applyMatrix4(globeGroup.matrixWorld);
      const ndc = worldPos.clone().project(camera);
      return {
        x: Number.isFinite(ndc.x) ? (ndc.x * 0.5 + 0.5) * width : width * 0.5,
        y: Number.isFinite(ndc.y) ? (-ndc.y * 0.5 + 0.5) * height : height * 0.5,
        z: Number.isFinite(worldPos.z) ? worldPos.z : 0
      };
    };

    let frameId;
    let clock = 0;
    const animate = () => {
      frameId = requestAnimationFrame(animate);
      clock += 0.015;
      const cfg = globeConfigRef.current || {};
      const speed = typeof cfg.rotationSpeed === 'number' ? cfg.rotationSpeed : 0.0032;
      const palette =
        AVATAR_CUSTOMIZATION_OPTIONS.globeThemes[cfg.theme] ||
        AVATAR_CUSTOMIZATION_OPTIONS.globeThemes.navy;

      if (autoRotateRef.current && !dragRef.current.isDragging) {
        rotationRef.current.yaw += speed;
      }

      const safeZoom = Number.isFinite(zoomRef.current) ? zoomRef.current : 1;
      globeGroup.scale.setScalar(safeZoom);
      globeGroup.rotation.y = rotationRef.current.yaw;
      globeGroup.rotation.x = rotationRef.current.pitch;
      globeGroup.updateMatrixWorld(true);

      ctx.save();
      ctx.scale(dpr, dpr);
      ctx.clearRect(0, 0, width, height);

      const cx = width / 2;
      const cy = height / 2;
      const edgePoint = project3DToScreen(new THREE.Vector3(1.23, 0, 0));
      const centerPoint = project3DToScreen(new THREE.Vector3(0, 0, 0));
      const rawDist = Math.hypot(edgePoint.x - centerPoint.x, edgePoint.y - centerPoint.y);
      const pixelRadius = Number.isFinite(rawDist) && rawDist > 10 ? Math.max(85, rawDist) : 130;

      if (cfg.showPulseRings !== false) {
        for (let rIdx = 1; rIdx <= 2; rIdx++) {
          const pulseR = pixelRadius + rIdx * 15 + Math.sin(clock * 2 + rIdx) * 3;
          ctx.beginPath();
          ctx.arc(cx, cy, pulseR, 0, Math.PI * 2);
          ctx.strokeStyle =
            rIdx % 2 === 0 ? 'rgba(243, 109, 137, 0.22)' : 'rgba(234, 184, 75, 0.26)';
          ctx.lineWidth = 1.4;
          ctx.stroke();
        }
      }

      const sphereGrad = ctx.createRadialGradient(
        cx - pixelRadius * 0.3,
        cy - pixelRadius * 0.3,
        pixelRadius * 0.1,
        cx,
        cy,
        pixelRadius
      );
      sphereGrad.addColorStop(0, palette.coreInner);
      sphereGrad.addColorStop(0.65, palette.coreMid);
      sphereGrad.addColorStop(1, palette.coreOuter);

      ctx.beginPath();
      ctx.arc(cx, cy, pixelRadius, 0, Math.PI * 2);
      ctx.fillStyle = sphereGrad;
      ctx.fill();
      ctx.lineWidth = 2;
      ctx.strokeStyle = 'rgba(234, 184, 75, 0.45)';
      ctx.stroke();

      for (let i = 0; i < dotItems.length; i++) {
        const item = dotItems[i];
        const pt = project3DToScreen(item.localPos);
        if (pt.z < 0.05) continue;
        const depthAlpha = Math.min(1, Math.max(0.15, (pt.z - 0.05) / 1.15));
        ctx.globalAlpha = depthAlpha;
        ctx.fillStyle =
          item.mask > 0.35
            ? palette.landPrimary
            : item.mask > -0.08
            ? palette.landSecondary
            : palette.oceanDot;
        ctx.beginPath();
        ctx.arc(pt.x, pt.y, item.size * safeZoom, 0, Math.PI * 2);
        ctx.fill();
      }
      ctx.restore();

      const nextProjected = membersRef.current.map((m) => {
        const localPos = sphericalToVec3(m.phi, m.theta, 1.28);
        const pt = project3DToScreen(localPos);
        const isFront = pt.z > -0.12;
        return {
          ...m,
          screenX: pt.x,
          screenY: pt.y,
          zDepth: pt.z,
          isFront
        };
      });
      setProjectedNodes(nextProjected);
    };
    animate();

    window.addEventListener('resize', syncCanvasSize);
    return () => {
      cancelAnimationFrame(frameId);
      window.removeEventListener('resize', syncCanvasSize);
    };
  }, []);

  return (
    <div className="relative rounded-[32px] border-2 overflow-hidden select-none bg-[radial-gradient(circle_at_center,#FFFDF9_0%,#EFE6D5_88%)] border-[#102F50] shadow-[0_10px_0_#102F50]">
      <div className="px-4 pt-3 flex flex-wrap items-center justify-between gap-2 text-[11px] font-extrabold text-[#102F50]">
        <span className="px-3 py-1 rounded-full bg-[#102F50] text-[#EAB84B]">
          🌍 Interactive 3D Family Globe · Drag to Rotate
        </span>
        <span className="text-[#D95732]">Tap any family member pin for live status &amp; call opt-in</span>
      </div>

      <div
        className="relative w-full h-[350px] sm:h-[395px] cursor-grab active:cursor-grabbing touch-none"
        onPointerDown={(e) => {
          dragRef.current = { isDragging: true, lastX: e.clientX, lastY: e.clientY };
        }}
        onPointerMove={(e) => {
          if (!dragRef.current.isDragging) return;
          const dx = e.clientX - dragRef.current.lastX;
          const dy = e.clientY - dragRef.current.lastY;
          dragRef.current.lastX = e.clientX;
          dragRef.current.lastY = e.clientY;
          rotationRef.current.yaw += dx * 0.007;
          rotationRef.current.pitch = Math.max(-0.6, Math.min(0.6, rotationRef.current.pitch + dy * 0.007));
        }}
        onPointerUp={() => {
          dragRef.current = { isDragging: false };
        }}
        onPointerLeave={() => {
          dragRef.current = { isDragging: false };
        }}
      >
        <div ref={mountRef} className="absolute inset-0" />

        {projectedNodes.map((node) => {
          if (!node.isFront) return null;
          const isNotifExpanded = expandedNotifId === node.id;
          const showAboveNotif = isNotifExpanded || node.hasNotification;

          return (
            <div
              key={node.id}
              style={{
                left: `${node.screenX}px`,
                top: `${node.screenY}px`,
                zIndex: Math.round((node.zDepth + 2) * 10),
                transform: 'translate(-50%, -50%)'
              }}
              className="absolute transition-transform duration-75"
            >
              {showAboveNotif && (
                <div style={{ bottom: 'calc(100% + 8px)' }} className="absolute left-1/2 -translate-x-1/2 z-40">
                  {isNotifExpanded ? (
                    <div className="w-56 p-3 rounded-2xl bg-[#102F50] text-[#FAF7F2] border-2 border-[#EAB84B] shadow-2xl">
                      <div className="flex items-center justify-between gap-1 mb-1">
                        <span className="px-2 py-0.5 rounded-full text-[9px] font-black uppercase bg-[#F36D89] text-[#102F50]">
                          {node.stickerBadge || 'Update'}
                        </span>
                        <button
                          onClick={(e) => {
                            e.stopPropagation();
                            setExpandedNotifId(null);
                          }}
                        >
                          <XIcon className="w-3 h-3" />
                        </button>
                      </div>
                      <p className="text-[11px] font-bold leading-snug">&ldquo;{node.statusText}&rdquo;</p>
                      <div className="mt-2 pt-2 border-t border-white/20 space-y-1.5 text-[10px]">
                        <div className="flex items-center justify-between">
                          <span className="text-[#EAB84B]">Call Opt-In:</span>
                          <span className="font-extrabold uppercase px-1.5 py-0.5 rounded bg-white/15">
                            {node.callOptIn || 'pending'}
                          </span>
                        </div>
                        {onUpdateMemberOptIn && (
                          <div className="flex gap-1 pt-1">
                            <button
                              onClick={(e) => {
                                e.stopPropagation();
                                onUpdateMemberOptIn(node.id, 'confirmed');
                              }}
                              className="flex-1 py-1 rounded bg-[#2E7D32] text-white font-black text-[9px]"
                            >
                              ✓ Confirm
                            </button>
                            <button
                              onClick={(e) => {
                                e.stopPropagation();
                                onUpdateMemberOptIn(node.id, 'ignored');
                              }}
                              className="flex-1 py-1 rounded bg-[#D95732] text-white font-black text-[9px]"
                            >
                              ✕ Ignore
                            </button>
                          </div>
                        )}
                      </div>
                    </div>
                  ) : (
                    <button
                      onClick={(e) => {
                        e.stopPropagation();
                        setExpandedNotifId(node.id);
                      }}
                      className="px-2.5 py-1 rounded-full bg-[#102F50] text-[#FAF7F2] border-2 border-[#F36D89] text-[10px] font-black shadow-lg whitespace-nowrap flex items-center gap-1"
                    >
                      <span className="w-2 h-2 rounded-full bg-[#F36D89] animate-ping" />
                      <span>{node.stickerBadge}</span>
                    </button>
                  )}
                </div>
              )}

              <div
                onClick={(e) => {
                  e.stopPropagation();
                  asmrAudio.playSoftChime(1.1);
                  setExpandedNotifId(expandedNotifId === node.id ? null : node.id);
                }}
                className="relative flex flex-col items-center cursor-pointer hover:scale-105 transition-transform"
              >
                <EditorialAvatar member={node} size="sm" />
                <span className="mt-1 px-2 py-0.5 rounded-full text-[10px] font-extrabold bg-[#FAF7F2] text-[#102F50] border border-[#102F50] shadow-sm whitespace-nowrap">
                  {node.name}
                </span>
              </div>
            </div>
          );
        })}
      </div>

      {/* Bottom Globe Controls & Member Quick Focus Pills */}
      <div className="relative z-20 flex flex-wrap items-center justify-between gap-2 px-4 pb-3 pt-1">
        <div className="flex flex-wrap gap-1">
          {members.map((m) => (
            <button
              key={m.id}
              onClick={() => {
                rotationRef.current.yaw = -m.theta;
                setExpandedNotifId(m.id);
                asmrAudio.playSoftChime(1.15);
              }}
              className={`px-2.5 py-1 rounded-full text-[10px] font-extrabold border transition-all ${
                expandedNotifId === m.id
                  ? 'bg-[#102F50] text-[#EAB84B] border-[#102F50]'
                  : 'bg-[#FAF7F2] text-[#102F50] border-[#DFD3BE] hover:border-[#102F50]'
              }`}
            >
              {m.name}
            </button>
          ))}
        </div>

        <div className="flex items-center gap-1.5">
          <button
            onClick={() => setZoom((z) => Math.max(0.75, +(z - 0.12).toFixed(2)))}
            className="w-7 h-7 rounded-full bg-white text-[#102F50] border border-[#102F50] flex items-center justify-center"
            title="Zoom Out"
          >
            <MinusIcon className="w-3.5 h-3.5" />
          </button>
          <button
            onClick={() => setZoom((z) => Math.min(1.35, +(z + 0.12).toFixed(2)))}
            className="w-7 h-7 rounded-full bg-white text-[#102F50] border border-[#102F50] flex items-center justify-center"
            title="Zoom In"
          >
            <PlusIcon className="w-3.5 h-3.5" />
          </button>
          <button
            onClick={() => setAutoRotate((r) => !r)}
            className={`w-7 h-7 rounded-full border border-[#102F50] flex items-center justify-center ${
              autoRotate ? 'bg-[#102F50] text-[#EAB84B]' : 'bg-white text-[#102F50]'
            }`}
            title="Toggle Auto-Rotate"
          >
            <RotateCcwIcon className="w-3.5 h-3.5" />
          </button>
        </div>
      </div>
    </div>
  );

    setFlippedIds(nextFlipped);

        setTimeout(() => {
          setCards((prev) =>
            prev.map((c, i) => (i === first || i === second ? { ...c, matched: true } : c))

            prev.map((c, i) => (i === first || i === second ? { ...c, flipped: false } : c))

            { id: 'bubbles', label: '🫧 Chatak-Patak Spice Pop' },
            { id: 'memory', label: '🃏 Keepsake Memory Match' },
            { id: 'bowl', label: '🥣 432Hz Singing Bowl' },
            { id: 'barnaby', label: '🐱 Barnaby the Cat' }

                activeGameTab === tab.id

                  : 'bg-[#FFFDF9] text-[#102F50] border-[#DFD3BE]'

            <p className="text-xs font-bold text-[#102F50]">
              Tap the warm spice bubbles for instant Web-Audio ASMR pops ({bubbles.filter((b) => b.popped).length}/{bubbles.length} popped)

            <button
